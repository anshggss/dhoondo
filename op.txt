Depth: 9, Type: TextNode, Data: func traverseNode(n *html.Node, depth int) {
	if n == nil {
		return
	}

	indent := strings.Repeat("  ", depth)

	switch n.Type {
	case html.ElementNode:
		// Print opening tag with attributes
		attrs := ""
		for _, attr := range n.Attr {
			attrs += fmt.Sprintf(" %s=\"%s\"", attr.Key, attr.Val)
		}
		fmt.Printf("%s<%s%s>\n", indent, n.Data, attrs)

	case html.TextNode:
		text := strings.TrimSpace(n.Data)
		if text != "" {
			fmt.Printf("%s%s\n", indent, text)
		}

	case html.CommentNode:
		fmt.Printf("%s<!-- %s -->\n", indent, n.Data)

	case html.DoctypeNode:
		fmt.Printf("%s<!DOCTYPE %s>\n", indent, n.Data)
	}

	// Traverse children
	for c := n.FirstChild; c != nil; c = c.NextSibling {
		traverseNode(c, depth+1)
	}
}

type Article struct {

func scrapeArticles(url string) ([]Article, error) {
	resp, err := http.Get(url)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	doc, err := html.Parse(resp.Body)
	if err != nil {
		return nil, err
	}

	var articles []Article
	var crawl func(*html.Node)
	
	crawl = func(n *html.Node) {
		if n.Type == html.ElementNode && n.Data == "article" {
			article := extractArticle(n)
			if article.Title != "" {
				articles = append(articles, article)
			}
		}

		for c := n.FirstChild; c != nil; c = c.NextSibling {
			crawl(c)
		}
	}

	crawl(doc)
	return articles, nil
}

func extractArticle(n *html.Node) Article {
	var article Article

	var findTitle func(*html.Node)
	findTitle = func(n *html.Node) {
		if n.Type == html.ElementNode {
			if n.Data == "h2" || n.Data == "h3" {
				article.Title = extractText(n)
			} else if n.Data == "a" {
				for _, attr := range n.Attr {
					if attr.Key == "href" {
						article.URL = attr.Val
					}
				}
			}
		}

		for c := n.FirstChild; c != nil; c = c.NextSibling {
			findTitle(c)
		}
	}

	findTitle(n)
	return article
}

func extractText(n *html.Node) string {
	if n.Type == html.TextNode {
		return n.Data
	}
	var text string
	for c := n.FirstChild; c != nil; c = c.NextSibling {
		text += extractText(c)
	}
	return text
}
